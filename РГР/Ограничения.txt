Ограничения по полям

User -- +Back 
	. fio -- min: 3, max: 255;
	. login-- min: 3, max: 255;
	. password -- min: 6, max: 255;

Customer -- +Back
	. masterID -- ;
	. FIO -- min: 2, max: 255, validator;
	. phone -- validator;
	. email -- validator;

Settings -- +Back +Front (Не нуждается в валидации на стороне пользователя)
	. userID -- ;
	. theme -- light|dark;
	. language -- ru|en;

Task -- +Back
	. customer_id -- ;
	. master_id -- ;
	. service -- ;
	. datetime -- datetime;
	. cost -- min: 500, max: 5000;

Weekend -- +Back
	. master_id -- ;
	. date -- date from datetime;
